public abstract class Product {
    private double price;
    private String name, category;
    private int ID, stock;


//constructor
    public Product(String name, int ID, double price, String category, int stock){
        this.name = name;
        this.ID = ID;
        this.price = price;
        this.category = category;
        this.stock = stock;

    }


    //getters and setters
    public String getName() {
        return name;
    }

    public int getID() {
        return ID;
    }

    public double getPrice() {
        return price;
    }

    public String getCategory() {
        return category;
    }

    public int getStock() {
        return stock;
    }

    public void setName(String name) {
        this.name = name;
    }

    public void setID(int ID) {
        this.ID = ID;
    }

    public void setPrice(double price) {
        this.price = price;
    }

    public void setCategory(String category) {
        this.category = category;
    }

    public void setStock(int stock) {
        this.stock = stock;
    }

    //special methods

    //displays complete information on said product
    public void displayFullInfo(){
        System.out.println("Product: " + this.name + "   Product ID: " + this.ID + "\nCategory: " + this.category + "\nPrice: $" + String.format("%.2f", this.price) + "\nTotal stock: " + this.stock);
    }

    //displays limited info on said product when browsing items
    public void listing(){
        if (this.checkStock()) {
            System.out.println(this.name + " $" + this.price + " is available | total in stock: " + this.stock);
        }else{
            System.out.println(this.name + " $" + this.price + " is unavailable");
        }
    }

    //checks if item is in stock
    public boolean checkStock(){
        return this.stock > 0;
    }
}
