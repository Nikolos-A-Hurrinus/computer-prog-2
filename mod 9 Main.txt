/*
this program essentially works as a small medical database that contains doctors, medical assistants and patients.
patients are assigned a doctor and the doctors are assigned patients.
each class is built off of a person class with the doctor class extending from the medical assistant class.
 */

public class Main {
    public static void main(String[] args) {
        //sets up doctors
        Doctor d1 = new Doctor("James", "Connor", 'A', 34, "04/07/91", "Main location", "Surgeon");
        Doctor d2 = new Doctor("Marcus", "Jones", 'J', 53, "03/17/72", "Satellite");

        //sets up medical assistants
        MedAssistant m1 = new MedAssistant("Jamie", "Longo", 'N', 24, "03/31/01");
        MedAssistant m2 = new MedAssistant("Alex", "Mason", 'D', 32, "10/14/93");
        MedAssistant m3 = new MedAssistant("Mary", "Flores", 'C', 50, "08/15/75", "Satellite");

        //sets up patients
        Patient p1 = new Patient("Brianna", "Gunning", 'P', 19, "06/21/06");
        Patient p2 = new Patient("Garrett", "Dunn", 'J', 77, "11/12/48", "12/15/24", "04/17/25", d1);
        Patient p3 = new Patient("Liam", "Chester", 'K', 15, "09/27/10", "02/14/25", "TBD", d2);
        Patient p4 = new Patient("Jolly", "Lopez", 'B', 44, "01/28/81", "03/14/25", "06/11/25", d2);

        //assigns doctors their patients
        Patient[] pa1 = {p2};
        d1.setPatients(pa1);

        Patient[] pa2 = {p3, p4};
        d2.setPatients(pa2);

        //displays everyone and their information
        System.out.println("Doctors:");
        d1.printInfo();
        d2.printInfo();

        System.out.println("\nMedical assistants:");
        m1.printInfo();
        m2.printInfo();
        m3.printInfo();

        System.out.println("\nPatients");
        p1.printInfo();
        p2.printInfo();
        p3.printInfo();
        p4.printInfo();
    }
}
