public class Grocery extends Product{
    private String expoDate;
    private double weight;

    public Grocery(String name, int ID, double pricePerPound, int stock, String expoDate, double weight) {
        super(name, ID, pricePerPound, "grocery", stock);
        this.expoDate = expoDate;
        this.weight = weight;
        this.setPrice(pricePerPound * weight);
    }

    //getters and setters
    public String getExpoDate() {
        return expoDate;
    }

    public double getWeight() {
        return weight;
    }

    public void setExpoDate(String expoDate) {
        this.expoDate = expoDate;
    }

    public void setWeight(double weight) {
        this.weight = weight;
    }

    //overrides the method to display all the info
    @Override
    public void displayFullInfo() {
        super.displayFullInfo();
        System.out.println("Expires: " + this.expoDate + "\nWeight: " + this.weight + " pounds");
    }
}
