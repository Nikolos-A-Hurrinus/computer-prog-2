public class Fib {

    public static int displayFibR(int n) {
        if (n <= 0) {
            System.out.println("can't be less than 1");
            return -1;
        } else if (n == 1) {
            return 0;
        } else if (n == 2) {
            return 1;
        } else {
            return displayFibR(n - 1) + displayFibR(n - 2);
        }
    }

    public static int displayFibI(int n) {
        if (n <= 0) {
            System.out.println("can't be less than 1");
            return -1;
        } else if (n == 1) {
            return 0;
        } else if (n == 2) {
            return 1;
        } else {
            int first = 0, second = 1;
            for (int i = 2; i < n; i++) {
                int temp = first + second;
                first = second;
                second = temp;
            }
            return second;
        }
    }

}
