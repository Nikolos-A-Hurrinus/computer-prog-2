public class Clothing extends Product{
    private String size, gender;


    public Clothing(String name, int ID, double price, int stock, String size, String gender) {
        super(name, ID, price, "clothing", stock);
        this.size = size.toUpperCase();
        this.gender = gender;
    }

    //getters setters
    public String getSize() {
        return size;
    }

    public String getGender() {
        return gender;
    }

    public void setSize(String size) {
        this.size = size;
    }

    public void setGender(String gender) {
        this.gender = gender;
    }


    //overrides the method to display all the info
    @Override
    public void displayFullInfo() {
        super.displayFullInfo();
        System.out.println("Size: " + this.size + " in " + this.gender);
    }


}
