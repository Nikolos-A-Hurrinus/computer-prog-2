import javax.swing.*;
import java.awt.*;
import java.awt.event.ActionEvent;
import java.awt.event.ActionListener;

public class Main {
    public static void main(String[] args) {

        //create frame
        JFrame f = new JFrame("registration");
        f.setSize(500,500);
        f.setResizable(false);
        f.getContentPane().setBackground(new Color(136, 132, 255));

        //set layout
        f.setLayout(new GridLayout(5,2));

        //make labels
        JLabel l1 = new JLabel("Name:");
        JLabel l2 = new JLabel("Email:");
        JLabel l3 = new JLabel("Password:");
        JLabel l4 = new JLabel("Confirm Password:");

        //make text boxes
        JTextField t1 = new JTextField();
        JTextField t2 = new JTextField();
        JTextField t3 = new JTextField();
        JTextField t4 = new JTextField();

        //make buttons
        JButton b1 = new JButton("Submit");
        JButton b2 = new JButton("Clear");

        //actions

        //submit action
        b1.addActionListener(new ActionListener() {
            public void actionPerformed(ActionEvent e) {
                if(t1.getText().isEmpty() || t2.getText().isEmpty() || t3.getText().isEmpty() || t4.getText().isEmpty()){
                    JOptionPane.showMessageDialog(f,"all text boxes must be filled out", "Insufficient Text", JOptionPane.INFORMATION_MESSAGE);
                } else if (!t4.getText().equals(t3.getText())) {
                    JOptionPane.showMessageDialog(f,"Password and Confirm Password do not match!", "Incorrect Password Match", JOptionPane.INFORMATION_MESSAGE);
                }else{
                    //creates dialog box for info to be displayed
                    JDialog d1 = new JDialog(f,"Information");
                    d1.setSize(350,200);
                    d1.setLayout(new GridLayout(3,1));

                    //creates labels for dialog box
                    JLabel bl1 = new JLabel("Name: " + t1.getText());
                    JLabel bl2 = new JLabel("Email: " + t2.getText());
                    JLabel bl3 = new JLabel("Password: " + t3.getText());

                    //adds labels to dialog box
                    d1.add(bl1);
                    d1.add(bl2);
                    d1.add(bl3);

                    //shows dialog box
                    d1.setVisible(true);
                }
            }
        });

        //clear action
        b2.addActionListener(new ActionListener() {
            public void actionPerformed(ActionEvent e) {
                t1.setText("");
                t2.setText("");
                t3.setText("");
                t4.setText("");

            }
        });

        //adds everything to frame
        f.add(l1);
        f.add(t1);
        f.add(l2);
        f.add(t2);
        f.add(l3);
        f.add(t3);
        f.add(l4);
        f.add(t4);
        f.add(b1);
        f.add(b2);

        //display frame
        f.setVisible(true);

        //exit JFrame
        f.setDefaultCloseOperation(JFrame.EXIT_ON_CLOSE);
    }
}
